async function api(path,body,key){
  const response=await fetch(path,{method:body===undefined?"GET":"POST",headers:{"Content-Type":"application/json",...(csrf?{"X-CSRF-Token":csrf}:{}),...(key?{"Idempotency-Key":key}:{})},body:body===undefined?undefined:JSON.stringify(body),cache:"no-store",credentials:"same-origin",signal:AbortSignal.timeout(15000)});
  const data=await response.json();
  if(!response.ok){if(response.status===401)locked();throw new Error(data.error??"Request failed");}return data;
}

function planView(project,opened=false){
  const plan=el("details",undefined,"plan-view"),summary=el("summary","Tasks, estimates and dependencies");plan.append(summary);
  plan.open=opened;
  if(project.goal)plan.append(el("p",project.goal,"plan-goal"));
  const byId=new Map(project.tasks.map(t=>[t.task_id,t])),levels=new Map(),visiting=new Set();
  function level(id){if(levels.has(id))return levels.get(id);if(visiting.has(id))return 0;visiting.add(id);const task=byId.get(id);const depth=task?1+Math.max(0,...task.depends_on.map(level)):0;visiting.delete(id);levels.set(id,depth);return depth;}
  for(const task of project.tasks)level(task.task_id);
  const total=project.tasks.reduce((sum,t)=>sum+t.estimate_minutes,0);
  const longest=new Map();function critical(id){if(longest.has(id))return longest.get(id);const task=byId.get(id);if(!task)return 0;const value=task.estimate_minutes+Math.max(0,...task.depends_on.map(critical));longest.set(id,value);return value;}
  const path=Math.max(0,...project.tasks.map(t=>critical(t.task_id)));
  plan.append(el("p",`${project.tasks.length} tasks · ${total} min authored effort · ${path} min dependency critical path. Estimates exclude review and retries.`,"plan-estimate"));
  const graph=el("div",undefined,"plan-graph");graph.setAttribute("aria-label","Task dependency graph by step");
  for(const depth of [...new Set(levels.values())].sort((a,b)=>a-b)){
    const column=el("div",undefined,"plan-layer");column.append(el("h3",`Step ${depth}`));
    for(const task of project.tasks.filter(t=>levels.get(t.task_id)===depth)){
      const card=el("div",undefined,"plan-node");card.append(el("strong",task.title),el("small",`${task.estimate_minutes} min · ${task.depends_on.length?`after ${task.depends_on.join(", ")}`:"starts independently"}`));column.append(card);
    }graph.append(column);
  }plan.append(graph);
  for(const task of project.tasks){const detail=el("details",undefined,"task-detail");detail.append(el("summary",`${task.title} · ${task.state} · ${task.estimate_minutes} min`),el("p",task.description||"No description projected."),el("p",`Depends on: ${task.depends_on.join(", ")||"nothing"}`));if(task.acceptance_criteria?.length)detail.append(el("p",`Acceptance: ${task.acceptance_criteria.join("; ")}`));plan.append(detail);}
  return plan;
}
